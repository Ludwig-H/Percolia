\begin{posterbox}{2}{\textsc{Hartigan}'s model and hierarchy}{33.5cm}
For a density $f$, clusters are the \textbf{connected components} of its superlevel sets \refb{1}.
\formulabox{$\displaystyle E_\lambda=\{y\in\mathbb R^d:f(y)\geq\lambda\}$}
Lowering $\lambda$ makes components grow and merge: a \textbf{hierarchy}.\par\vspace{0.75cm}
\vfill
\centering\resizebox{0.87\linewidth}{!}{\input{figures/hartigan.tex}}\par
\vfill
\RaggedRight We seek to \textbf{estimate the unknown density $f$}. A natural, non-parametric choice is the \textbf{\mbox{$K$-nearest-neighbour} estimator} \refb{2}:
\formulabox{$\displaystyle\widehat f\gets\widehat f_{K\text{-NN}}$}
\formulabox{$\displaystyle\widehat f_{K\text{-NN}}(y)=\frac{K}{n\,\omega_d\,\textcolor{inriablue}{r_K(y)}^d}$}
\smalltext{\textcolor{inriablue}{\textbf{$r_K(y)$}}: distance to the $K$-th nearest sample point.}
\captiontext{$n$: number of points; $\omega_d$: unit-ball volume.}
\vfill
\textbf{Change of variable: density level to neighbour radius.}
\formulabox{{\fontsize{28}{33}\selectfont$\displaystyle\widehat f_{K\text{-NN}}(y)\geq\lambda\ \Longleftrightarrow\ \textcolor{inriablue}{r_K(y)}\leq\bigl(K/(n\omega_d\lambda)\bigr)^{1/d}=:r(\lambda)$}}
\end{posterbox}
\gap
