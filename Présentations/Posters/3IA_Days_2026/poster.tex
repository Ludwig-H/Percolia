\begin{posterbox}{1}{What is clustering?}{19.3cm}
\textbf{Group similar observations without class labels.}
Density-based clustering can recover non-convex groups and leave noisy points unassigned.
\vfill
\centering\resizebox{\linewidth}{!}{\input{figures/clustering.tex}}\par
\vfill
\captiontext{2D example: HDBSCAN \refb{3}. Colours indicate clusters; grey points remain unassigned.}
\vspace{0.2cm}
\RaggedRight A partition gives \emph{groups}; a hierarchy also tells us \textbf{at which scale they merge}.
\end{posterbox}
\gap
\begin{posterbox}{2}{The mathematical model}{28cm}
For a probability density $f$ on $\mathbb{R}^d$, Hartigan's clusters are the \textbf{connected components} of a density superlevel set \refb{1}:
\[
 E_\lambda=\{y\in\mathbb{R}^d:f(y)\geq\lambda\}.
\]
As $\lambda$ decreases, components appear, grow and merge.
\vfill
\centering\resizebox{\linewidth}{!}{\input{figures/hartigan.tex}}\par
\vfill
\RaggedRight For $X=\{x_1,\ldots,x_n\}$, estimate density from the distance $r_K(y)$ to the $K$-th nearest sample point \refb{2}:
\[
 \widehat f_K(y)=\frac{K}{n\,\omega_d\,r_K(y)^d}.
\]
\smalltext{$\omega_d$ is the volume of the unit ball; $K$ controls the local density scale.}
\vspace{0.2cm}
\begin{beamercolorbox}[sep=0.45cm,wd=\linewidth]{block example body}
\centering\fontsize{31}{36}\selectfont
\textbf{High density $\Longleftrightarrow$ small neighbour radius}\\[0.2cm]
$\{\widehat f_K\geq\lambda\}=\{r_K\leq r\}$,
\quad $r=\bigl(K/(n\omega_d\lambda)\bigr)^{1/d}$.
\end{beamercolorbox}
\end{posterbox}
\gap
\begin{posterbox}{3}{HGP: connect at the right order}{30cm}
The region covered by at least $K$ closed balls is
\[
 E_{K,r}=\{y:\#(X\cap B(y,r))\geq K\}
       =\bigcup_{\substack{S\subseteq X\\|S|=K}}\;\bigcap_{x\in S}B(x,r).
\]
\textbf{Replace edges by higher-order interactions.}
In the \v{C}ech complex, connect $(K-1)$-simplices through $K$-simplices: their components are \textbf{$K$-polyhedra} \myref{4,5}.
\vfill
\centering\resizebox{!}{12.0cm}{\input{figures/overlaps.tex}}\par
\vfill
\captiontext{$K=2$ on an equilateral triangle of side $a$: the three pairwise overlaps merge at $r=a/\sqrt{3}$. Connectivity is \emph{through triangles}, not a shared vertex.}
\vspace{0.35cm}
\begin{beamercolorbox}[sep=0.45cm,wd=\linewidth]{block example body}
\fontsize{31}{36}\selectfont
\textbf{Exact correspondence.} The $K$-polyhedra recover the connected components of $E_{K,r}$ at every radius \myref{4}.
\end{beamercolorbox}
\vspace{0.3cm}
\RaggedRight\smalltext{\textbf{HGP-Clusterer} uses geometric sparsification to process critical merges. $K=1$ gives Single-Linkage (minimum spanning tree); for $K\geq2$, clusters may share points \myref{4}.}
\end{posterbox}

\textbf{Polyhedra as tokens; the hierarchy as multiscale context.}
Use edges, triangles and tetrahedra to build geometric components, rather than treating all points independently.
\vfill
\centering\resizebox{0.73\linewidth}{!}{\input{figures/polyhedral_hierarchy.tex}}\par
\vfill
\centering\resizebox{\linewidth}{!}{\input{figures/learning_pipeline.tex}}\par

\vspace{0.25cm}
\par\smalltext{\textcolor{inriared}{\textbf{Research hypothesis:}} this geometric structure may improve learned 3D representations. It remains to be tested against the same backbone without hierarchical guidance.}
\end{posterbox}
