\begin{posterbox}{2}{Hartigan's density model}{22.5cm}
\textbf{Clusters are connected components} of density superlevel sets \refb{1}:
\[
 E_\lambda=\{y\in\mathbb R^d:f(y)\geq\lambda\}.
\]
\centering\resizebox{\linewidth}{!}{\input{figures/hartigan.tex}}\par
\captiontext{Lower $\lambda$: components grow and merge.}
\vfill
\RaggedRight\textbf{$K$-nearest-neighbour density estimate} \refb{2}:
\[
 \widehat f_K(y)=\frac{K}{n\,\omega_d\,r_K(y)^d}.
\]
\captiontext{$r_K(y)$: distance to the $K$-th sample point; $\omega_d$: unit-ball volume. High density means small $r_K$.}
\end{posterbox}
